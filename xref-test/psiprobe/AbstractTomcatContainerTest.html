128   @Test
129   void setWrapperInitializesContainerStateAndRemovesValveOnUnset() {
130     TestTomcatContainer container = new TestTomcatContainer();
131     Wrapper wrapper = mock(Wrapper.class);
132     Container parent = mock(Container.class);
133     Host host = mock(Host.class);
134     Engine engine = mock(Engine.class);
135     Service service = mock(Service.class);
136     Pipeline pipeline = mock(Pipeline.class);
137 
138     when(wrapper.getParent()).thenReturn(parent);
139     when(parent.getParent()).thenReturn(host);
140     when(host.getParent()).thenReturn(engine);
141     when(host.getPipeline()).thenReturn(pipeline);
142     when(host.getName()).thenReturn("localhost");
143     when(host.getAppBase()).thenReturn("webapps");
144     when(engine.getService()).thenReturn(service);
145     when(engine.getName()).thenReturn("Catalina");
146     when(service.findConnectors()).thenReturn(new Connector[] {mock(Connector.class)});
147 
148     container.setWrapper(wrapper);
149     assertEquals(1, container.findConnectors().size());
150 
151     container.setWrapper(null);
152     verify(pipeline, times(1)).addValve(any());
153     verify(pipeline, times(1)).removeValve(any());
154   }

185   @Test
186   void getAppBaseSupportsRelativeAndAbsolutePaths() {
187     TestTomcatContainer container = new TestTomcatContainer();
188     Host host = mock(Host.class);
189 
190     container.host = host;
191 
192     String oldBase = System.getProperty("catalina.base");
193     try {
194       System.setProperty("catalina.base", "/tmp/catalina-base");
195 
196       when(host.getAppBase()).thenReturn("webapps");
197       assertEquals(Path.of("/tmp/catalina-base", "webapps").toString(),
198           container.getAppBase().toPath().toString());
199 
200       String absoluteAppBase =
201           Path.of(System.getProperty("java.io.tmpdir"), "opt-webapps").toAbsolutePath().toString();
202       when(host.getAppBase()).thenReturn(absoluteAppBase);
203       assertEquals(Path.of(absoluteAppBase).toString(), container.getAppBase().toPath().toString());
204     } finally {
205       if (oldBase != null) {
206         System.setProperty("catalina.base", oldBase);
207       } else {
208         System.clearProperty("catalina.base");
209       }
210     }
211   }
