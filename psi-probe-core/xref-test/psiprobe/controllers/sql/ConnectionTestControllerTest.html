40  class ConnectionTestControllerTest {
41  
42    private ConnectionTestController controller;
43    private ContainerWrapperBean containerWrapper;
44    private ResourceResolver resourceResolver;
45    private TomcatContainer tomcatContainer;
46  
47    @BeforeEach
48    void setUp() throws Exception {
49      controller = new ConnectionTestController();
50      controller.setViewName("ajax/sql/connection");
51  
52      StaticApplicationContext applicationContext = new StaticApplicationContext();
53      applicationContext.addMessage("probe.src.dataSourceTest.resource.lookup.failure",
54          Locale.getDefault(), "lookup {0}");
55      applicationContext.addMessage("probe.src.dataSourceTest.connection.failure",
56          Locale.getDefault(), "connection {0}");
57      applicationContext.addMessage("probe.jsp.dataSourceTest.dbMetaData.dbProdName",
58          Locale.getDefault(), "db product");
59      applicationContext.addMessage("probe.jsp.dataSourceTest.dbMetaData.dbProdVersion",
60          Locale.getDefault(), "db version");
61      applicationContext.addMessage("probe.jsp.dataSourceTest.dbMetaData.jdbcDriverName",
62          Locale.getDefault(), "driver name");
63      applicationContext.addMessage("probe.jsp.dataSourceTest.dbMetaData.jdbcDriverVersion",
64          Locale.getDefault(), "driver version");
65      applicationContext.addMessage("probe.jsp.dataSourceTest.dbMetaData.jdbcVersion",
66          Locale.getDefault(), "jdbc version");
67      applicationContext.refresh();
68      controller.setApplicationContext(applicationContext);
69  
70      containerWrapper = mock(ContainerWrapperBean.class);
71      resourceResolver = mock(ResourceResolver.class);
72      tomcatContainer = mock(TomcatContainer.class);
73  
74      when(containerWrapper.getResourceResolver()).thenReturn(resourceResolver);
75      when(containerWrapper.getTomcatContainer()).thenReturn(tomcatContainer);
76      when(tomcatContainer.formatContextName("app")).thenReturn("/app");
77  
78      controller.setContainerWrapper(containerWrapper);
79    }
80  
81    @Test
82    void handleRequestReturnsDatabaseMetadataWhenLookupSucceeds() throws Exception {
83      Context context = mock(Context.class);
84      DataSource dataSource = mock(DataSource.class);
85      Connection connection = mock(Connection.class);
86      DatabaseMetaData metaData = mock(DatabaseMetaData.class);
87  
88      when(tomcatContainer.findContext("/app")).thenReturn(context);
89      when(resourceResolver.lookupDataSource(context, "jdbc/test", containerWrapper))
90          .thenReturn(dataSource);
91      when(dataSource.getConnection()).thenReturn(connection);
92      when(connection.getMetaData()).thenReturn(metaData);
93      when(metaData.getDatabaseProductName()).thenReturn("PostgreSQL");
94      when(metaData.getDatabaseProductVersion()).thenReturn("16");
95      when(metaData.getDriverName()).thenReturn("Driver");
96      when(metaData.getDriverVersion()).thenReturn("1.0");
97      when(metaData.getJDBCMajorVersion()).thenReturn(4);
98  
99      MockHttpServletRequest request = new MockHttpServletRequest("GET", "/sql/connection.ajax");
100     request.addParameter("webapp", "app");
101     request.addParameter("resource", "jdbc/test");
102 
103     ModelAndView modelAndView = controller.handleRequest(request, new MockHttpServletResponse());
104 
105     assertEquals("ajax/sql/connection", modelAndView.getViewName());
106     @SuppressWarnings("unchecked")
107     List<Map<String, String>> dbMetaData =
108         (List<Map<String, String>>) modelAndView.getModel().get("dbMetaData");
109     assertEquals(5, dbMetaData.size());
110     assertEquals("db product", dbMetaData.get(0).get("propertyName"));
111     assertEquals("PostgreSQL", dbMetaData.get(0).get("propertyValue"));
112   }
113 
114   @Test
115   void handleRequestSetsLookupErrorWhenDatasourceLookupFails() throws Exception {
116     Context context = mock(Context.class);
117     when(tomcatContainer.findContext("/app")).thenReturn(context);
118     when(resourceResolver.lookupDataSource(context, "jdbc/missing", containerWrapper))
119         .thenThrow(new NamingException("boom"));
120 
121     MockHttpServletRequest request = new MockHttpServletRequest("GET", "/sql/connection.ajax");
122     request.addParameter("webapp", "app");
123     request.addParameter("resource", "jdbc/missing");
124 
125     ModelAndView modelAndView = controller.handleRequest(request, new MockHttpServletResponse());
126 
127     assertEquals("ajax/sql/connection", modelAndView.getViewName());
128     assertEquals("lookup jdbc/missing", request.getAttribute("errorMessage"));
129   }
130 
131   @Test
132   void handleRequestSetsConnectionErrorWhenConnectionFails() throws Exception {
133     Context context = mock(Context.class);
134     DataSource dataSource = mock(DataSource.class);
135     when(tomcatContainer.findContext("/app")).thenReturn(context);
136     when(resourceResolver.lookupDataSource(context, "jdbc/test", containerWrapper))
137         .thenReturn(dataSource);
138     when(dataSource.getConnection()).thenThrow(new SQLException("down"));
139 
140     MockHttpServletRequest request = new MockHttpServletRequest("GET", "/sql/connection.ajax");
141     request.addParameter("webapp", "app");
142     request.addParameter("resource", "jdbc/test");
143 
144     ModelAndView modelAndView = controller.handleRequest(request, new MockHttpServletResponse());
145 
146     assertEquals("ajax/sql/connection", modelAndView.getViewName());
147     assertEquals("connection down", request.getAttribute("errorMessage"));
148     assertTrue(modelAndView.getModel().isEmpty());
149   }
